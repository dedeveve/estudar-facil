        function mostrarConteudo(chave) {
            const caixa = document.getElementById('conteudo-estudo');
            const dados = conteudos[chave] || {
                titulo: 'Conteúdo em Desenvolvimento',
                matematica: conteudoPadrao.matematica,
                portugues: conteudoPadrao.portugues,
                ciencias: null
            };

            document.getElementById('titulo-serie').textContent = dados.titulo;
            document.getElementById('conteudo-matematica').innerHTML = dados.matematica;
            document.getElementById('conteudo-portugues').innerHTML = dados.portugues;

            const caixaCiencias = document.getElementById('caixa-ciencias');
            if (dados.ciencias) {
                caixaCiencias.classList.remove('hidden');
                document.getElementById('conteudo-ciencias').innerHTML = dados.ciencias;
            } else {
                caixaCiencias.classList.add('hidden');
            }

            caixa.classList.remove('hidden');
            caixa.scrollIntoView({ behavior: 'smooth' });
        }

        function voltarEscolha() {
            document.getElementById('conteudo-estudo').classList.add('hidden');
            document.getElementById('escolher-ano').scrollIntoView({ behavior: 'smooth' });
